The repo

A regex crossword solver written in Python. It parses each row and column regex with a PLY grammar (see grammar.md), turns the puzzle into an SMT problem, and solves it with Z3. Backreferences are supported, so puzzles that use groups and references such as \1 can be solved, including rectangular pu

class RegexLexer:
    tokens = (
        "NON_META_CHAR", "DIGIT",
        "DOT", "STAR", "PLUS", "QUESTION",
        "BAR",
        "LPAREN", "RPAREN",
        "LBRACKET", "RBRACKET",
        "CARET", "DASH", "DOLLAR",
        "LBRACE", "RBRACE", "COMMA",
        "BACKSLASH",
    )

    def t_NON_META_CHAR(self, t):
        r"[a-zA-Z :/'!]"
        return t
    def t_DIGIT(self, t):
        r"[0-9]"
        return t
    t_DOT, t_STAR, t_PLUS, t_QUESTION = r"\.", r"\*", r"\+", r"\?"
    t_BAR = r"\|"
